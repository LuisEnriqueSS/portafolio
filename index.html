class TenantContextResolver:
    # Clean Architecture en acción
    def resolve_tenant_context(self, request: Request) -> TenantContext:
        tenant_id = request.headers.get("X-Tenant-ID")

        if not tenant_id:
            raise TenantIdentifierMissing()

        tenant = self.tenant_repo.find_by_id(tenant_id)
        self.db_manager.set_schema(tenant.schema_name)

        return TenantContext(tenant_id, tenant.config)
